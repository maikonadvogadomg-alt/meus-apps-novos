  <script>
    let supabase = null;
    let currentUser = null;

    const $ = (id) => document.getElementById(id);

    async function initSupabase() {
      const url = localStorage.getItem("supabase_url");
      const key = localStorage.getItem("supabase_key");

      if (!url || !key) return false;

      supabase = window.supabase.createClient(url, key);

      const { data: { user } } = await supabase.auth.getUser();
      currentUser = user;

      if (!currentUser) return false;

      await loadConfig();
      return true;
    }

    async function loadConfig() {
      if (!supabase || !currentUser) return;

      const { data } = await supabase
        .from("user_configs")
        .select("*")
        .eq("user_id", currentUser.id)
        .single();

      if (data) {
        $("openaiKey").value = data.openai_key || "";
        $("githubToken").value = data.github_token || "";
        $("googleTtsKey").value = data.google_tts_key || "";
      }
    }

    async function saveConfig() {
      if (!supabase || !currentUser) return;

      const payload = {
        user_id: currentUser.id,
        openai_key: $("openaiKey").value,
        github_token: $("githubToken").value,
        google_tts_key: $("googleTtsKey").value
      };

      const { data: existing } = await supabase
        .from("user_configs")
        .select("id")
        .eq("user_id", currentUser.id)
        .single();

      let result;

      if (existing) {
        result = await supabase
          .from("user_configs")
          .update(payload)
          .eq("user_id", currentUser.id);
      } else {
        result = await supabase
          .from("user_configs")
          .insert(payload);
      }

      if (result.error) {
        showStatus("Erro: " + result.error.message, "error");
        return;
      }

      showStatus("Configuração salva com sucesso!", "success");
    }

    function showStatus(message, type = "success") {
      const el = $("status");
      el.className = "status show " + type;
      el.textContent = message;

      setTimeout(() => {
        el.className = "status";
      }, 3000);
    }

    $("saveBtn").addEventListener("click", saveConfig);

    $("clearBtn").addEventListener("click", () => {
      if (confirm("Limpar todas as configurações?")) {
        $("openaiKey").value = "";
        $("githubToken").value = "";
        $("googleTtsKey").value = "";
        saveConfig();
      }
    });

    initSupabase().then(success => {
      if (!success) {
        showStatus("Erro ao conectar", "error");
      }
    });
  </script>

          .from("snippets")
          .delete()
          .eq("id", project.id);

          return;

        await loadCloudProjects();
        showStatus("success", "Projeto excluído do Supabase.");
        return;

      state.localProjects = state.localProjects.filter((p) => p.id !== project.id);
      localStorage.setItem(LOCAL_PROJECTS_KEY, JSON.stringify(state.localProjects));
      renderProjects();
      showStatus("success", "Projeto local excluído.");

          .from("snippets")
          .update({ title: title.trim() })
          .eq("id", project.id)
          .then(() => loadCloudProjects());
        return;

      if (local) {

    async function saveCloud() {
      if (!state.supabase && !connectSupabase()) {
        showStatus("error", "Configure Supabase antes de salvar.");
        return;
      }

      const payload = {
        title: state.title.trim() || "Sem título",
        html: state.html,
        css: state.css,
        js: state.js
      };

      let result;

      if (state.currentId && state.currentId.length > 20) {
        result = await state.supabase
          .from("snippets")
          .update(payload)
          .eq("id", state.currentId)
          .select("id")
          .single();
      } else {
        result = await state.supabase
          .from("snippets")
          .insert(payload)
          .select("id")
          .single();
      }

      if (result.error) {
        showStatus("error", "Erro ao salvar: " + result.error.message);
        return;
      }

      state.currentId = result.data.id;
      persistState();
      await loadCloudProjects();
      showStatus("success", "Projeto salvo no Supabase.");
    }

    async function loadCloudProjects() {
      if (!state.supabase) return;

      const { data, error } = await state.supabase
        .from("snippets")
        .select("id,title,html,css,js")
        .order("title", { ascending: true });

      if (error) {
        showStatus("error", "Erro ao carregar: " + error.message);
        return;
      }

      state.cloudProjects = (data || []).map((p) => ({
        ...p,
        source: "cloud"
      }));

      renderProjects();
    }

      const cloud = state.cloudProjects.map((p) => ({ ...p, source: "cloud" }));

    }

    function downloadFile(filename, content, type) {
      const blob = new Blob([content], { type });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");

      link.href = url;
      link.download = filename;
      document.body.appendChild(link);
      link.click();
      link.remove();

      setTimeout(() => URL.revokeObjectURL(url), 1000);
    }

    function importFile(file) {
      const reader = new FileReader();

      reader.onload = () => {
        const content = String(reader.result || "");
        const ext = file.name.split(".").pop().toLowerCase();

        state.title = file.name.replace(/\.[^.]+$/, "") || "Importado";

        if (ext === "css") {
          state.tab = "css";
          state.css = content;
        } else if (ext === "js") {
          state.tab = "js";
          state.js = content;
        } else {
          state.tab = "html";
          state.html = content;
        }

        state.currentId = null;
        persistState();
        renderAll();
        updatePreview();
        showStatus("success", "Arquivo importado.");
      };

      reader.readAsText(file);
    }

    function parseCsv(text) {
      const rows = [];
      let row = [];
      let value = "";
      let quoted = false;

      text = text.replace(/^\uFEFF/, "");

      for (let i = 0; i < text.length; i++) {
        const char = text[i];
        const next = text[i + 1];

        if (char === '"' && quoted && next === '"') {
          value += '"';
          i++;
        } else if (char === '"') {
          quoted = !quoted;
        } else if (char === "," && !quoted) {
          row.push(value);
          value = "";
        } else if ((char === "\n" || char === "\r") && !quoted) {
          if (char === "\r" && next === "\n") i++;
          row.push(value);
          if (row.some((item) => item.trim())) rows.push(row);
          row = [];
          value = "";
        } else {
          value += char;
        }
      }

      if (value || row.length) {
        row.push(value);
        rows.push(row);
      }

      if (rows.length < 2) return [];

      const headers = rows[0].map((h) => h.trim().toLowerCase());

      return rows.slice(1).map((cols) => {
        const result = {};
        headers.forEach((h, i) => {
          result[h] = cols[i] || "";
        });
        return result;
      });
    }

    async function importCsvFile(file) {
      const text = await file.text();
      const rows = parseCsv(text);

      if (!rows.length) {
        showStatus("error", "CSV vazio ou inválido.");
        return;
      }

      let imported = 0;

      for (const row of rows) {
        const project = {
          id: crypto.randomUUID(),
          title: row.title || "Importado",
          html: row.html || "",
          css: row.css || "",
          js: row.js || "",
          saved_at: new Date().toISOString()
        };

        state.localProjects.unshift(project);
        imported++;
      }

      localStorage.setItem(LOCAL_PROJECTS_KEY, JSON.stringify(state.localProjects));
      renderProjects();
      showStatus("success", imported + " projeto(s) importado(s).");
    }

    function parseCsv(text) {
      const rows = [];
      let row = [];
      let value = "";
      let quoted = false;

      text = text.replace(/^\uFEFF/, "");

      for (let i = 0; i < text.length; i++) {
        const char = text[i];
        const next = text[i + 1];

        if (char === '"' && quoted && next === '"') {
          value += '"';
          i++;
        } else if (char === '"') {
          quoted = !quoted;
        } else if (char === "," && !quoted) {
          row.push(value);
          value = "";
        } else if ((char === "\n" || char === "\r") && !quoted) {
          if (char === "\r" && next === "\n") i++;
          row.push(value);
          if (row.some((item) => item.trim())) rows.push(row);
          row = [];
          value = "";
        } else {
          value += char;
        }
      }

      if (value || row.length) {
        row.push(value);
        rows.push(row);
      }

      if (rows.length < 2) return [];

      const headers = rows[0].map((h) => h.trim().toLowerCase());

      return rows.slice(1).map((cols) => {
        const result = {};
        headers.forEach((h, i) => {
          result[h] = cols[i] || "";
        });
        return result;
      });
    }

    async function importCsvFile(file) {
      const text = await file.text();
      const rows = parseCsv(text);

      if (!rows.length) {
        showStatus("error", "CSV vazio ou inválido.");
        return;
      }

      let imported = 0;

      for (const row of rows) {
        const project = {
          id: crypto.randomUUID(),
          title: row.title || "Importado",
          html: row.html || "",
          css: row.css || "",
          js: row.js || "",
          saved_at: new Date().toISOString()
        };

        state.localProjects.unshift(project);
        imported++;
      }

      localStorage.setItem(LOCAL_PROJECTS_KEY, JSON.stringify(state.localProjects));
      renderProjects();
      showStatus("success", imported + " projeto(s) importado(s).");
    }

    let supabase = null;
    let currentUser = null;

    const state = {
      mode: "html",
      tab: "html",

      css: DEFAULT_CSS,

    async function initSupabase() {
      const url = localStorage.getItem("supabase_url");
      const key = localStorage.getItem("supabase_key");

      if (!url || !key) {
        showLoginStatus("Configure Supabase primeiro.", "error");
        return false;
      }

      supabase = window.supabase.createClient(url, key);
      return true;
    }

    async function handleLogin() {
      const email = $("loginEmail").value.trim();
      const password = $("loginPassword").value.trim();

      if (!email || !password) {
        showLoginStatus("Preencha email e senha.", "error");
        return;
      }

      if (!await initSupabase()) return;

      const { data, error } = await supabase.auth.signInWithPassword({
        email,
        password
      });

      if (error) {
        showLoginStatus("Erro: " + error.message, "error");
        return;
      }

      currentUser = data.user;
      showLoginStatus("Bem-vindo!", "success");
      setTimeout(showApp, 1000);
    }

    async function handleSignup() {
      const email = $("loginEmail").value.trim();
      const password = $("loginPassword").value.trim();

      if (!email || !password) {
        showLoginStatus("Preencha email e senha.", "error");
        return;
      }

      if (!await initSupabase()) return;

      const { data, error } = await supabase.auth.signUp({
        email,
        password
      });

      if (error) {
        showLoginStatus("Erro: " + error.message, "error");
        return;
      }

      showLoginStatus("Cadastro realizado! Verifique seu email.", "success");
    }

    function showLoginStatus(message, type) {
      const el = $("loginStatus");
      el.className = "login-status show " + type;
      el.textContent = message;
    }

    function showApp() {
      $("loginScreen").classList.add("hidden");
      $("appScreen").style.display = "flex";
      $("userEmail").textContent = currentUser?.email || "Usuário";
      loadState();
      updatePreview();
      loadCloudProjects();
    }

    async function handleLogout() {
      if (!supabase) return;

      await supabase.auth.signOut();
      currentUser = null;
      $("loginScreen").classList.remove("hidden");
      $("appScreen").style.display = "none";
      $("loginEmail").value = "";
      $("loginPassword").value = "";
    }

    // ===== UTILITÁRIOS =====

    function showStatus(type, message, timeout = 4000) {
      const el = $("status");
      el.className = "status show " + type;
      el.textContent = message;

      if (timeout) {
        clearTimeout(showStatus.timer);
        showStatus.timer = setTimeout(() => {
          el.className = "status";
        }, timeout);
      }
    }

    function showConfigStatus(message, type = "") {
      const el = $("configStatus");
      el.textContent = message;
      if (type === "success") el.style.color = "#86efac";
      else if (type === "error") el.style.color = "#fca5a5";
      else el.style.color = "inherit";
    }

    function safeJsonParse(value, fallback) {
      try {
        return JSON.parse(value);
      } catch {
        return fallback;
      }
    }

    // ===== PERSISTÊNCIA =====

    function loadState() {
      const saved = safeJsonParse(localStorage.getItem(STORAGE_KEY), null);
      if (saved) {
        Object.assign(state, saved);
      }

      state.localProjects = safeJsonParse(
        localStorage.getItem(LOCAL_PROJECTS_KEY),
        []
      );

      renderAll();
    }

    function persistState() {
      const copy = {
        mode: state.mode,
        tab: state.tab,
        title: state.title,
        html: state.html,
        css: state.css,
        js: state.js,
        currentId: state.currentId,
        autoRun: state.autoRun,
        editorCollapsed: state.editorCollapsed,
        mobileView: state.mobileView
      };

      localStorage.setItem(STORAGE_KEY, JSON.stringify(copy));
    }

    // ===== EDITOR =====

    function getCurrentCode() {
      return state[state.tab] || "";
    }

    function setCurrentCode(value) {
      state[state.tab] = value;
      persistState();
      updateEditor();
      if (state.autoRun) {
        schedulePreview();
      }
    }

    function updateEditor() {
      const value = getCurrentCode();
      $("codeEditor").value = value;

      const lines = Math.max(1, value.split("\n").length);
      $("lineNumbers").textContent = Array.from(
        { length: lines },
        (_, i) => i + 1
      ).join("\n");

      $("projectTitle").value = state.title;
    }

    function setTab(tab) {
      state.tab = tab;
      persistState();
      renderTabs();
      updateEditor();
    }

    function renderTabs() {
      document.querySelectorAll("[data-tab]").forEach((btn) => {
        btn.classList.toggle("active", btn.dataset.tab === state.tab);
      });
    }

    // ===== PREVIEW =====

    function buildDocument() {
      if (state.html.toLowerCase().includes("<!doctype")) {
        return state.html;
      }

      return `<!DOCTYPE html>
<html lang="pt-BR">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <style>${state.css}</style>
</head>
<body>
  ${state.html}
  <script>
try {
${state.js}
} catch (error) {
  document.body.insertAdjacentHTML("beforeend",
    "<pre style='color:red;padding:12px;white-space:pre-wrap'>" +
    error.stack +
    "</pre>"
  );
}
  <\/script>
</body>
</html>`;
    }

    let previewTimer = null;

    function schedulePreview() {
      clearTimeout(previewTimer);
      previewTimer = setTimeout(updatePreview, 300);
    }

    function updatePreview() {
      persistState();

      const blob = new Blob([buildDocument()], { type: "text/html;charset=utf-8" });
      const url = URL.createObjectURL(blob);
      const oldUrl = $("previewFrame").dataset.objectUrl;

      $("previewFrame").src = url;
      $("previewFrame").dataset.objectUrl = url;

      if (oldUrl) {
        setTimeout(() => URL.revokeObjectURL(oldUrl), 1000);
      }
    }

    // ===== PROJETOS =====

    function getProjectRecord() {
      return {
        id: state.currentId || crypto.randomUUID(),
        title: state.title.trim() || "Sem título",
        html: state.html,
        css: state.css,
        js: state.js,
        saved_at: new Date().toISOString()
      };
    }

    function saveLocal() {
      const record = getProjectRecord();
      const idx = state.localProjects.findIndex((p) => p.id === state.currentId);

      if (idx >= 0) {
        state.localProjects[idx] = record;
      } else {
        state.currentId = record.id;
        state.localProjects.unshift(record);
      }

      localStorage.setItem(LOCAL_PROJECTS_KEY, JSON.stringify(state.localProjects));
      persistState();
      renderProjects();
      showStatus("success", "Projeto salvo localmente.");
    }

    async function saveCloud() {
      if (!supabase || !currentUser) {
        showStatus("error", "Faça login primeiro.");
        return;
      }

      const payload = {
        user_id: currentUser.id,
        title: state.title.trim() || "Sem título",
        html: state.html,
        css: state.css,
        js: state.js
      };

      let result;

      if (state.currentId && state.currentId.length > 20) {
        result = await supabase
          .from("snippets")
          .update(payload)
          .eq("id", state.currentId)
          .select("id")
          .single();
      } else {
        result = await supabase
          .from("snippets")
          .insert(payload)
          .select("id")
          .single();
      }

      if (result.error) {
        showStatus("error", "Erro ao salvar: " + result.error.message);
        return;
      }

      state.currentId = result.data.id;
      persistState();
      await loadCloudProjects();
      showStatus("success", "Projeto salvo no Supabase!");
    }

    async function loadCloudProjects() {
      if (!supabase || !currentUser) return;

      const { data, error } = await supabase
        .from("snippets")
        .select("id,title,html,css,js")
        .eq("user_id", currentUser.id)
        .order("created_at", { ascending: false });

      if (error) {
        console.error("Erro ao carregar projetos:", error);
        return;
      }

      state.cloudProjects = (data || []).map((p) => ({
        ...p,
        source: "cloud"
      }));

      renderProjects();
    }

    function loadProject(project) {
      state.currentId = project.id;
      state.title = project.title || "Sem título";
      state.html = project.html || "";
      state.css = project.css || "";
      state.js = project.js || "";
      state.tab = "html";

      persistState();
      renderAll();
      updatePreview();
      closeOpenModal();
      showStatus("success", "Projeto aberto!");
    }

    async function deleteProject(project) {
      if (!confirm('Excluir "' + project.title + '"?')) return;

      if (project.source === "cloud" && supabase) {
        const { error } = await supabase
          .from("snippets")
          .delete()
          .eq("id", project.id);

        if (error) {
          showStatus("error", "Erro: " + error.message);
          return;
        }

        await loadCloudProjects();
        showStatus("success", "Projeto excluído.");
        return;
      }

      state.localProjects = state.localProjects.filter((p) => p.id !== project.id);
      localStorage.setItem(LOCAL_PROJECTS_KEY, JSON.stringify(state.localProjects));
      renderProjects();
      showStatus("success", "Projeto local excluído.");
    }

    function renameProject(project) {
      const title = prompt("Novo nome:", project.title);
      if (!title || !title.trim()) return;

      if (project.source === "cloud" && supabase) {
        supabase
          .from("snippets")
          .update({ title: title.trim() })
          .eq("id", project.id)
          .then(() => loadCloudProjects());
        return;
      }

      const local = state.localProjects.find((p) => p.id === project.id);
      if (local) {
        local.title = title.trim();
        localStorage.setItem(LOCAL_PROJECTS_KEY, JSON.stringify(state.localProjects));
        renderProjects();
      }
    }

    // ===== RENDERIZAÇÃO =====

    function mergedProjects() {
      const cloud = state.cloudProjects.map((p) => ({ ...p, source: "cloud" }));
      const local = state.localProjects.map((p) => ({ ...p, source: "local" }));
      return [...cloud, ...local];
    }

    function renderProjects() {
      const term = $("searchInput").value.trim().toLowerCase();
      const projects = mergedProjects().filter((p) =>
        String(p.title || "").toLowerCase().includes(term)
      );

      $("projectList").innerHTML = "";

      if (!projects.length) {
        $("projectList").innerHTML = '<div class="empty">Nenhum projeto.</div>';
        return;
      }

      projects.forEach((p, idx) => {
        $("projectList").appendChild(createProjectElement(p, idx + 1));
      });

      renderOpenProjects();
    }

    function createProjectElement(project, number) {
      const item = document.createElement("div");
      item.className = "project-item";

      const info = document.createElement("div");
      info.className = "project-info";

      const name = document.createElement("div");
      name.className = "project-name";
      name.textContent = "#" + number + " — " + (project.title || "Sem título");

      const meta = document.createElement("div");
      meta.className = "project-meta";
      meta.textContent = (project.source === "cloud" ? "☁ Supabase" : "💾 Local");

      info.append(name, meta);
      info.addEventListener("click", () => loadProject(project));

      const actions = document.createElement("div");
      actions.className = "project-actions";

      const rename = document.createElement("button");
      rename.className = "small-btn";
      rename.textContent = "✎";
      rename.addEventListener("click", (e) => {
        e.stopPropagation();
        renameProject(project);
      });

      const del = document.createElement("button");
      del.className = "small-btn";
      del.textContent = "×";
      del.addEventListener("click", (e) => {
        e.stopPropagation();
        deleteProject(project);
      });

      actions.append(rename, del);
      item.append(info, actions);

      return item;
    }

    function renderOpenProjects() {
      const term = $("openSearchInput").value.trim().toLowerCase();
      const container = $("openProjectList");
      container.innerHTML = "";

      const projects = mergedProjects().filter((p) =>
        String(p.title || "").toLowerCase().includes(term)
      );

      if (!projects.length) {
        container.innerHTML = '<div class="empty">Nenhum projeto.</div>';
        return;
      }

      projects.forEach((p, idx) => {
        container.appendChild(createProjectElement(p, idx + 1));
      });
    }

    function renderLayout() {
      $("editorPanel").classList.toggle("collapsed", state.editorCollapsed);
      $("expandEditorBtn").style.display = state.editorCollapsed ? "inline-flex" : "none";
      $("collapseBtn").style.display = state.editorCollapsed ? "none" : "inline-flex";

      $("autoRunBtn").classList.toggle("primary", state.autoRun);
      $("autoRunBtn").textContent = state.autoRun ? "◉ Ao vivo" : "○ Manual";

      const isMobile = window.innerWidth <= 600;

      if (isMobile) {
        $("editorPanel").classList.toggle("mobile-hidden", state.mobileView !== "editor");
        $("previewPanel").classList.toggle("mobile-hidden", state.mobileView !== "preview");
      } else {
        $("editorPanel").classList.remove("mobile-hidden");
        $("previewPanel").classList.remove("mobile-hidden");
      }
    }

    function renderAll() {
      renderTabs();
      updateEditor();
      renderLayout();
      renderProjects();
    }

      $("openModal").classList.add("open");
      renderOpenProjects();

      $("openModal").classList.remove("open");

    function downloadFile(filename, content, type) {
      const blob = new Blob([content], { type });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");

      link.href = url;
      link.download = filename;
      document.body.appendChild(link);
      link.click();
      link.remove();

      setTimeout(() => URL.revokeObjectURL(url), 1000);
    }

    function downloadCurrent() {
      const filename = (state.title.trim() || "playground")
        .replace(/[^\wÀ-ÿ -]/g, "")
        .replace(/\s+/g, "-") + ".html";

      downloadFile(filename, buildDocument(), "text/html;charset=utf-8");
      showStatus("success", "Arquivo baixado.");
    }

    async function copyCurrent() {
      try {
        await navigator.clipboard.writeText(buildDocument());
        showStatus("success", "Código copiado!");
      } catch {
        showStatus("error", "Não foi possível copiar.");
      }
    }

    function openFullscreen() {
      const blob = new Blob([buildDocument()], { type: "text/html;charset=utf-8" });
      const url = URL.createObjectURL(blob);
      window.open(url, "_blank", "noopener,noreferrer");
      setTimeout(() => URL.revokeObjectURL(url), 5000);
    }

    function importFile(file) {
      const reader = new FileReader();

      reader.onload = () => {
        const content = String(reader.result || "");
        const ext = file.name.split(".").pop().toLowerCase();

        state.title = file.name.replace(/\.[^.]+$/, "") || "Importado";

        if (ext === "css") {
          state.tab = "css";
          state.css = content;
        } else if (ext === "js") {
          state.tab = "js";
          state.js = content;
        } else {
          state.tab = "html";
          state.html = content;
        }

        state.currentId = null;
        persistState();
        renderAll();
        updatePreview();
        showStatus("success", "Arquivo importado.");
      };

      reader.readAsText(file);
    }

    function resetCurrent() {
      state.html = DEFAULT_HTML;
      state.css = DEFAULT_CSS;
      state.js = DEFAULT_JS;
      state.title = "Sem título";
      state.currentId = null;
      state.tab = "html";

      persistState();
      renderAll();
      updatePreview();
      showStatus("success", "Restaurado!");
    }

    function clearCurrent() {
      state.html = "";
      state.css = "";
      state.js = "";

      persistState();
      renderAll();
      updatePreview();
      showStatus("info", "Editor limpo.");
    }

    function toggleTheme() {
      const isLight = document.documentElement.classList.toggle("light");
      localStorage.setItem("playground_theme", isLight ? "light" : "dark");
      $("themeBtn").textContent = isLight ? "☀" : "☾";
    }

    function applyTheme() {
      const theme = localStorage.getItem("playground_theme") || "dark";
      document.documentElement.classList.toggle("light", theme === "light");
      $("themeBtn").textContent = theme === "light" ? "☀" : "☾";
    }

    function bindEvents() {
      // Login
      $("loginBtn").addEventListener("click", handleLogin);
      $("signupBtn").addEventListener("click", handleSignup);

      // Abas
      document.querySelectorAll("[data-tab]").forEach((btn) => {
        btn.addEventListener("click", () => setTab(btn.dataset.tab));
      });

      // Título
      $("projectTitle").addEventListener("input", (e) => {
        state.title = e.target.value;
        persistState();
      });

      // Editor
      $("codeEditor").addEventListener("input", (e) => {
        setCurrentCode(e.target.value);
      });

      // Botões
      $("autoRunBtn").addEventListener("click", () => {
        state.autoRun = !state.autoRun;
        persistState();
        renderLayout();
        if (state.autoRun) updatePreview();
      });

      $("runBtn").addEventListener("click", updatePreview);
      $("saveLocalBtn").addEventListener("click", saveLocal);
      $("saveCloudBtn").addEventListener("click", saveCloud);
      $("openBtn").addEventListener("click", openOpenModal);
      $("closeOpenBtn").addEventListener("click", closeOpenModal);
      $("copyBtn").addEventListener("click", copyCurrent);
      $("downloadBtn").addEventListener("click", downloadCurrent);
      $("fullscreenBtn").addEventListener("click", openFullscreen);
      $("resetBtn").addEventListener("click", resetCurrent);
      $("clearBtn").addEventListener("click", clearCurrent);
      $("themeBtn").addEventListener("click", toggleTheme);
      $("logoutBtn").addEventListener("click", handleLogout);

      // Collapse
      $("collapseBtn").addEventListener("click", () => {
        state.editorCollapsed = true;
        persistState();
        renderLayout();
      });

      $("expandEditorBtn").addEventListener("click", () => {
        state.editorCollapsed = false;
        persistState();
        renderLayout();
      });

      // Mobile
      $("mobileEditorBtn").addEventListener("click", () => {
        state.mobileView = "editor";
        renderLayout();
      });

      $("mobilePreviewBtn").addEventListener("click", () => {
        state.mobileView = "preview";
        updatePreview();
        renderLayout();
      });

      // Config
      $("configBtn").addEventListener("click", () => {
        $("configModal").classList.add("open");
      });

      $("closeConfigBtn").addEventListener("click", () => {
        $("configModal").classList.remove("open");
      });

      // Pesquisa
      $("searchInput").addEventListener("input", renderProjects);
      $("openSearchInput").addEventListener("input", renderOpenProjects);

      // Atualizar
      $("refreshBtn").addEventListener("click", () => {
        loadCloudProjects();
        showStatus("info", "Atualizado!");
      });

      // Import
      $("importBtn").addEventListener("click", () => $("fileInput").click());
      $("fileInput").addEventListener("change", (e) => {
        const file = e.target.files[0];
        if (file) importFile(file);
        e.target.value = "";
      });

      // Resize
      window.addEventListener("resize", renderLayout);
    }

    // ===== INICIALIZAÇÃO =====

    applyTheme();
    bindEvents();

  <script>
    let supabase = null;
    let currentUser = null;
    let currentSession = null;
    let userConfig = null;
    let tokenManager = new TokenManager();
    let brainSystem = null;

    const $ = (id) => document.getElementById(id);

    async function initSupabase() {
      const url = localStorage.getItem("supabase_url");
      const key = localStorage.getItem("supabase_key");

      if (!url || !key) return false;

      supabase = window.supabase.createClient(url, key);

      const { data: { user } } = await supabase.auth.getUser();
      currentUser = user;

      if (!currentUser) return false;

      await loadUserConfig();
      await loadSessions();

      return true;
    }

    async function loadUserConfig() {
      if (!supabase || !currentUser) return;

      const { data } = await supabase
        .from("user_configs")
        .select("*")
        .eq("user_id", currentUser.id)
        .single();

      userConfig = data;
    }

    async function loadSessions() {
      if (!supabase || !currentUser) return;

      const { data } = await supabase
        .from("chat_sessions")
        .select("*")
        .eq("user_id", currentUser.id)
        .order("created_at", { ascending: false });

      $("sessionsList").innerHTML = "";

      if (!data || data.length === 0) {
        $("sessionsList").innerHTML = '<div style="color: var(--muted); font-size: 11px;">Nenhuma conversa</div>';
        return;
      }

      data.forEach(session => {
        const item = document.createElement("div");
        item.className = "session-item";
        item.textContent = session.title;
        item.addEventListener("click", () => loadSession(session));
        $("sessionsList").appendChild(item);
      });
    }

    async function loadSession(session) {
      currentSession = session;
      brainSystem = new BrainSystem(supabase, session.id);

      await brainSystem.loadMemories();
      await brainSystem.loadCodeFragments();

      document.querySelectorAll(".session-item").forEach(item => {
        item.classList.remove("active");
      });

      event.target.classList.add("active");

      $("sessionTitle").textContent = session.title;

      const { data: messages } = await supabase
        .from("chat_messages")
        .select("*")
        .eq("session_id", session.id)
        .order("created_at", { ascending: true });

      $("chatMessages").innerHTML = "";

      if (messages && messages.length > 0) {
        messages.forEach(msg => {
          addMessageToUI(msg.role, msg.content);
        });
      }

      updateTokenDisplay();
    }

    async function createNewSession() {
      if (!supabase || !currentUser) return;

      const { data, error } = await supabase
        .from("chat_sessions")
        .insert({
          user_id: currentUser.id,
          title: "Conversa " + new Date().toLocaleTimeString(),
          model: tokenManager.currentModel,
          temperature: 0.7,
          max_tokens: tokenManager.getModelInfo().maxTokensPerRequest
        })
        .select()
        .single();

      if (error) {
        showStatus("Erro ao criar conversa", "error");
        return;
      }

      currentSession = data;
      brainSystem = new BrainSystem(supabase, data.id);
      tokenManager.reset();

      await loadSessions();
      await loadSession(data);
      showStatus("Nova conversa criada");
    }

    async function sendMessage() {
      const text = $("chatInput").value.trim();

      if (!text) return;

      if (!currentSession) {
        showStatus("Selecione uma conversa primeiro", "error");
        return;
      }

      if (!userConfig?.openai_key) {
        showStatus("Configure sua chave OpenAI", "error");
        return;
      }

      // Verificar tokens
      const estimatedTokens = tokenManager.estimateTokens(text);
      const canRequest = tokenManager.canMakeRequest(estimatedTokens);

      if (!canRequest.allowed) {
        showStatus("❌ " + canRequest.reason, "error");
        return;
      }

      // Construir prompt com contexto do cérebro
      let finalPrompt = brainSystem.buildContinuationPrompt(text, tokenManager);
      const model = tokenManager.getModelInfo();
      finalPrompt += brainSystem.buildFragmentationInstructions(model.maxCodeLines);

      // Adicionar mensagem do usuário
      addMessageToUI("user", text);

      await supabase.from("chat_messages").insert({
        session_id: currentSession.id,
        role: "user",
        content: text,
        tokens_used: estimatedTokens
      });

      $("chatInput").value = "";

      // Chamar IA
      try {
        showStatus("⏳ Aguardando resposta...", "");

        const response = await fetch("https://api.openai.com/v1/chat/completions", {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${userConfig.openai_key}`
          },
          body: JSON.stringify({
            model: currentSession.model,
            messages: [{ role: "user", content: finalPrompt }],
            max_tokens: currentSession.max_tokens,
            temperature: currentSession.temperature
          })
        });

        if (!response.ok) {
          const error = await response.json();
          throw new Error(error.error?.message || "Erro na API");
        }

        const data = await response.json();
        const aiMessage = data.choices[0].message.content;
        const inputTokens = data.usage.prompt_tokens;
        const outputTokens = data.usage.completion_tokens;

        addMessageToUI("assistant", aiMessage);

        // Extrair código
        const codeBlocks = brainSystem.extractCodeBlocks(aiMessage);

        // Salvar mensagem
        await supabase.from("chat_messages").insert({
          session_id: currentSession.id,
          role: "assistant",
          content: aiMessage,
          tokens_used: outputTokens
        });

        // Salvar no cérebro
        await brainSystem.saveMemory(text, aiMessage, codeBlocks);

        // Registrar uso
        const tokenData = tokenManager.addTokens(inputTokens, outputTokens);

        await supabase.from("ai_usage").insert({
          user_id: currentUser.id,
          session_id: currentSession.id,
          model: currentSession.model,
          tokens_input: inputTokens,
          tokens_output: outputTokens,
          tokens_total: tokenData.totalTokens,
          cost: tokenData.cost
        });

        updateTokenDisplay();
        showStatus("✅ Resposta recebida", "success");

      } catch (error) {
        showStatus("❌ Erro: " + error.message, "error");
        addMessageToUI("system", "Erro ao conectar com IA: " + error.message);
      }
    }

    function addMessageToUI(role, content) {
      const container = $("chatMessages");

      if (container.querySelector(".empty")) {
        container.innerHTML = "";
      }

      const msg = document.createElement("div");
      msg.className = "message " + role;

      // Formatar código
      if (role === "assistant") {
        msg.innerHTML = content
          .replace(/```(\w+)?\n([\s\S]*?)```/g, '<pre><code>$2</code></pre>')
          .replace(/`([^`]+)`/g, '<code>$1</code>');
      } else {
        msg.textContent = content;
      }

      container.appendChild(msg);
      container.scrollTop = container.scrollHeight;
    }

    function updateTokenDisplay() {
      const progress = tokenManager.getProgress();

      $("tokenFill").style.width = progress.dailyPercent + "%";
      $("tokenFill").className = "token-fill";

      if (progress.dailyPercent > 80) {
        $("tokenFill").classList.add("danger");
      } else if (progress.dailyPercent > 50) {
        $("tokenFill").classList.add("warning");
      }

      $("tokenText").textContent = `${progress.dailyUsed} / ${progress.dailyMax}`;
      $("modelInfo").textContent = progress.model;
      $("sessionTokens").textContent = progress.sessionUsed;

      // Calcular custo
      const cost = tokenManager.calculateCost(progress.sessionUsed);
      $("costInfo").textContent = "R$ " + (cost * 5.5).toFixed(2); // Conversão USD para BRL
    }

    function showStatus(message, type = "success") {
      const el = $("status");
      el.className = "status show " + type;
      el.textContent = message;

      if (type === "success") {
        setTimeout(() => {
          el.className = "status";
        }, 3000);
      }
    }

    // Eventos
    $("sendBtn").addEventListener("click", sendMessage);
    $("chatInput").addEventListener("keypress", (e) => {
      if (e.key === "Enter" && !e.shiftKey) {
        e.preventDefault();
        sendMessage();
      }
    });

    $("modelSelector").addEventListener("change", (e) => {
      tokenManager.setModel(e.target.value);
      updateTokenDisplay();
    });

    $("newSessionBtn").addEventListener("click", createNewSession);
    $("newSessionBtn2").addEventListener("click", createNewSession);

    $("deleteSessionBtn").addEventListener("click", async () => {
      if (!currentSession || !confirm("Deletar esta conversa?")) return;

      await supabase.from("chat_sessions").delete().eq("id", currentSession.id);
      currentSession = null;
      await loadSessions();
      showStatus("Conversa deletada");
    });

    $("exportBtn").addEventListener("click", async () => {
      if (!currentSession) return;

      const { data: messages } = await supabase
        .from("chat_messages")
        .select("*")
        .eq("session_id", currentSession.id)
        .order("created_at", { ascending: true });

      const { data: brain } = await supabase
        .from("brain_history")
        .select("*")
        .eq("session_id", currentSession.id)
        .order("iteration", { ascending: true });

      const { data: code } = await supabase
        .from("code_fragments")
        .select("*")
        .eq("session_id", currentSession.id)
        .order("fragment_order", { ascending: true });

      const exportData = {
        session: currentSession,
        messages,
        brain,
        code,
        exportedAt: new Date().toISOString()
      };

      const json = JSON.stringify(exportData, null, 2);
      const blob = new Blob([json], { type: "application/json" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = currentSession.title + "-completo.json";
      link.click();
      URL.revokeObjectURL(url);

      showStatus("Exportado com sucesso!");
    });

    // Inicializar
    initSupabase().then(success => {
      if (!success) {
        showStatus("Erro ao conectar", "error");
      } else {
        updateTokenDisplay();
      }
    });
  </script>

  <script>
    let supabase = null;
    let currentUser = null;
    let currentSession = null;
    let userConfig = null;
    let tokenManager = new TokenManager();
    let brainSystem = null;

    const $ = (id) => document.getElementById(id);

    async function initSupabase() {
      const url = localStorage.getItem("supabase_url");
      const key = localStorage.getItem("supabase_key");

      if (!url || !key) return false;

      supabase = window.supabase.createClient(url, key);

      const { data: { user } } = await supabase.auth.getUser();
      currentUser = user;

      if (!currentUser) return false;

      await loadUserConfig();
      await loadSessions();

      return true;
    }

    async function loadUserConfig() {
      if (!supabase || !currentUser) return;

      const { data } = await supabase
        .from("user_configs")
        .select("*")
        .eq("user_id", currentUser.id)
        .single();

      userConfig = data;
    }

    async function loadSessions() {
      if (!supabase || !currentUser) return;

      const { data } = await supabase
        .from("chat_sessions")
        .select("*")
        .eq("user_id", currentUser.id)
        .order("created_at", { ascending: false });

      $("sessionsList").innerHTML = "";

      if (!data || data.length === 0) {
        $("sessionsList").innerHTML = '<div style="color: var(--muted); font-size: 11px;">Nenhuma conversa</div>';
        return;
      }

      data.forEach(session => {
        const item = document.createElement("div");
        item.className = "session-item";
        item.textContent = session.title;
        item.addEventListener("click", () => loadSession(session));
        $("sessionsList").appendChild(item);
      });
    }

    async function loadSession(session) {
      currentSession = session;
      brainSystem = new BrainSystem(supabase, session.id);

      await brainSystem.loadMemories();
      await brainSystem.loadCodeFragments();

      document.querySelectorAll(".session-item").forEach(item => {
        item.classList.remove("active");
      });

      event.target.classList.add("active");

      $("sessionTitle").textContent = session.title;

      const { data: messages } = await supabase
        .from("chat_messages")
        .select("*")
        .eq("session_id", session.id)
        .order("created_at", { ascending: true });

      $("chatMessages").innerHTML = "";

      if (messages && messages.length > 0) {
        messages.forEach(msg => {
          addMessageToUI(msg.role, msg.content);
        });
      }

      updateTokenDisplay();
    }

    async function createNewSession() {
      if (!supabase || !currentUser) return;

      const { data, error } = await supabase
        .from("chat_sessions")
        .insert({
          user_id: currentUser.id,
          title: "Conversa " + new Date().toLocaleTimeString(),
          model: tokenManager.currentModel,
          temperature: 0.7,
          max_tokens: tokenManager.getModelInfo().maxTokensPerRequest
        })
        .select()
        .single();

      if (error) {
        showStatus("Erro ao criar conversa", "error");
        return;
      }

      currentSession = data;
      brainSystem = new BrainSystem(supabase, data.id);
      tokenManager.reset();

      await loadSessions();
      await loadSession(data);
      showStatus("Nova conversa criada");
    }

    async function sendMessage() {
      const text = $("chatInput").value.trim();

      if (!text) return;

      if (!currentSession) {
        showStatus("Selecione uma conversa primeiro", "error");
        return;
      }

      if (!userConfig?.openai_key) {
        showStatus("Configure sua chave OpenAI", "error");
        return;
      }

      // Verificar tokens
      const estimatedTokens = tokenManager.estimateTokens(text);
      const canRequest = tokenManager.canMakeRequest(estimatedTokens);

      if (!canRequest.allowed) {
        showStatus("❌ " + canRequest.reason, "error");
        return;
      }

      // Construir prompt com contexto do cérebro
      let finalPrompt = brainSystem.buildContinuationPrompt(text, tokenManager);
      const model = tokenManager.getModelInfo();
      finalPrompt += brainSystem.buildFragmentationInstructions(model.maxCodeLines);

      // Adicionar mensagem do usuário
      addMessageToUI("user", text);

      await supabase.from("chat_messages").insert({
        session_id: currentSession.id,
        role: "user",
        content: text,
        tokens_used: estimatedTokens
      });

      $("chatInput").value = "";

      // Chamar IA
      try {
        showStatus("⏳ Aguardando resposta...", "");

        const response = await fetch("https://api.openai.com/v1/chat/completions", {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${userConfig.openai_key}`
          },
          body: JSON.stringify({
            model: currentSession.model,
            messages: [{ role: "user", content: finalPrompt }],
            max_tokens: currentSession.max_tokens,
            temperature: currentSession.temperature
          })
        });

        if (!response.ok) {
          const error = await response.json();
          throw new Error(error.error?.message || "Erro na API");
        }

        const data = await response.json();
        const aiMessage = data.choices[0].message.content;
        const inputTokens = data.usage.prompt_tokens;
        const outputTokens = data.usage.completion_tokens;

        addMessageToUI("assistant", aiMessage);

        // Extrair código
        const codeBlocks = brainSystem.extractCodeBlocks(aiMessage);

        // Salvar mensagem
        await supabase.from("chat_messages").insert({
          session_id: currentSession.id,
          role: "assistant",
          content: aiMessage,
          tokens_used: outputTokens
        });

        // Salvar no cérebro
        await brainSystem.saveMemory(text, aiMessage, codeBlocks);

        // Registrar uso
        const tokenData = tokenManager.addTokens(inputTokens, outputTokens);

        await supabase.from("ai_usage").insert({
          user_id: currentUser.id,
          session_id: currentSession.id,
          model: currentSession.model,
          tokens_input: inputTokens,
          tokens_output: outputTokens,
          tokens_total: tokenData.totalTokens,
          cost: tokenData.cost
        });

        updateTokenDisplay();
        showStatus("✅ Resposta recebida", "success");

      } catch (error) {
        showStatus("❌ Erro: " + error.message, "error");
        addMessageToUI("system", "Erro ao conectar com IA: " + error.message);
      }
    }

    function addMessageToUI(role, content) {
      const container = $("chatMessages");

      if (container.querySelector(".empty")) {
        container.innerHTML = "";
      }

      const msg = document.createElement("div");
      msg.className = "message " + role;

      // Formatar código
      if (role === "assistant") {
        msg.innerHTML = content
          .replace(/```(\w+)?\n([\s\S]*?)```/g, '<pre><code>$2</code></pre>')
          .replace(/`([^`]+)`/g, '<code>$1</code>');
      } else {
        msg.textContent = content;
      }

      container.appendChild(msg);
      container.scrollTop = container.scrollHeight;
    }

    function updateTokenDisplay() {
      const progress = tokenManager.getProgress();

      $("tokenFill").style.width = progress.dailyPercent + "%";
      $("tokenFill").className = "token-fill";

      if (progress.dailyPercent > 80) {
        $("tokenFill").classList.add("danger");
      } else if (progress.dailyPercent > 50) {
        $("tokenFill").classList.add("warning");
      }

      $("tokenText").textContent = `${progress.dailyUsed} / ${progress.dailyMax}`;
      $("modelInfo").textContent = progress.model;
      $("sessionTokens").textContent = progress.sessionUsed;

      // Calcular custo
      const cost = tokenManager.calculateCost(progress.sessionUsed);
      $("costInfo").textContent = "R$ " + (cost * 5.5).toFixed(2); // Conversão USD para BRL
    }

    function showStatus(message, type = "success") {
      const el = $("status");
      el.className = "status show " + type;
      el.textContent = message;

      if (type === "success") {
        setTimeout(() => {
          el.className = "status";
        }, 3000);
      }
    }

    // Eventos
    $("sendBtn").addEventListener("click", sendMessage);
    $("chatInput").addEventListener("keypress", (e) => {
      if (e.key === "Enter" && !e.shiftKey) {
        e.preventDefault();
        sendMessage();
      }
    });

    $("modelSelector").addEventListener("change", (e) => {
      tokenManager.setModel(e.target.value);
      updateTokenDisplay();
    });

    $("newSessionBtn").addEventListener("click", createNewSession);
    $("newSessionBtn2").addEventListener("click", createNewSession);

    $("deleteSessionBtn").addEventListener("click", async () => {
      if (!currentSession || !confirm("Deletar esta conversa?")) return;

      await supabase.from("chat_sessions").delete().eq("id", currentSession.id);
      currentSession = null;
      await loadSessions();
      showStatus("Conversa deletada");
    });

    $("exportBtn").addEventListener("click", async () => {
      if (!currentSession) return;

      const { data: messages } = await supabase
        .from("chat_messages")
        .select("*")
        .eq("session_id", currentSession.id)
        .order("created_at", { ascending: true });

      const { data: brain } = await supabase
        .from("brain_history")
        .select("*")
        .eq("session_id", currentSession.id)
        .order("iteration", { ascending: true });

      const { data: code } = await supabase
        .from("code_fragments")
        .select("*")
        .eq("session_id", currentSession.id)
        .order("fragment_order", { ascending: true });

      const exportData = {
        session: currentSession,
        messages,
        brain,
        code,
        exportedAt: new Date().toISOString()
      };

      const json = JSON.stringify(exportData, null, 2);
      const blob = new Blob([json], { type: "application/json" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = currentSession.title + "-completo.json";
      link.click();
      URL.revokeObjectURL(url);

      showStatus("Exportado com sucesso!");
    });

    // Inicializar
    initSupabase().then(success => {
      if (!success) {
        showStatus("Erro ao conectar", "error");
      } else {
        updateTokenDisplay();
      }
    });
  </script>

// ===== BRAIN SYSTEM =====
// Mantém contexto mesmo com limite de tokens

class BrainSystem {
  constructor(supabase, sessionId) {
    this.supabase = supabase;
    this.sessionId = sessionId;
    this.maxMemories = 10; // Manter últimas 10 interações
    this.memories = [];
    this.codeFragments = [];
  }

  async loadMemories() {
    if (!this.supabase || !this.sessionId) return;

    const { data } = await this.supabase
      .from("brain_history")
      .select("*")
      .eq("session_id", this.sessionId)
      .order("iteration", { ascending: false })
      .limit(this.maxMemories);

    this.memories = (data || []).reverse();
  }

  async loadCodeFragments() {
    if (!this.supabase || !this.sessionId) return;

    const { data } = await this.supabase
      .from("code_fragments")
      .select("*")
      .eq("session_id", this.sessionId)
      .order("fragment_order", { ascending: true });

    this.codeFragments = data || [];
  }

  // Criar resumo do contexto para enviar à IA
  buildContextPrompt() {
    if (this.memories.length === 0) return "";

    let context = "## CONTEXTO ANTERIOR\n\n";

    this.memories.forEach((mem, idx) => {
      context += `### Interação ${idx + 1}\n`;
      context += `**Pergunta:** ${mem.context}\n`;
      context += `**Pontos-chave:** ${mem.key_points}\n\n`;
    });

    if (this.codeFragments.length > 0) {
      context += "## CÓDIGO ANTERIOR\n\n";
      this.codeFragments.forEach((frag, idx) => {
        context += `### Parte ${idx + 1} (${frag.language})\n`;
        context += "```" + frag.language + "\n";
        context += frag.code + "\n";
        context += "```\n\n";
      });
    }

    return context;
  }

  // Salvar nova memória
  async saveMemory(userMessage, aiResponse, codeBlocks = []) {
    if (!this.supabase || !this.sessionId) return;

    const iteration = this.memories.length + 1;

    // Salvar memória
    await this.supabase.from("brain_history").insert({
      session_id: this.sessionId,
      iteration,
      context: userMessage.substring(0, 500),
      code_snippets: codeBlocks.length > 0 ? JSON.stringify(codeBlocks) : null,
      key_points: aiResponse.substring(0, 500)
    });

    // Salvar fragmentos de código
    for (let i = 0; i < codeBlocks.length; i++) {
      await this.supabase.from("code_fragments").insert({
        session_id: this.sessionId,
        fragment_order: this.codeFragments.length + i + 1,
        language: codeBlocks[i].language || "javascript",
        code: codeBlocks[i].code,
        description: codeBlocks[i].description || ""
      });
    }

    await this.loadMemories();
    await this.loadCodeFragments();
  }

  // Extrair blocos de código da resposta da IA
  extractCodeBlocks(text) {
    const codeBlockRegex = /```(\w+)?\n([\s\S]*?)```/g;
    const blocks = [];
    let match;

    while ((match = codeBlockRegex.exec(text)) !== null) {
      blocks.push({
        language: match[1] || "javascript",
        code: match[2].trim()
      });
    }

    return blocks;
  }

  // Fragmentar código grande em partes menores
  fragmentCode(code, maxLines = 50) {
    const lines = code.split("\n");
    const fragments = [];

    for (let i = 0; i < lines.length; i += maxLines) {
      fragments.push(lines.slice(i, i + maxLines).join("\n"));
    }

    return fragments;
  }

  // Gerar prompt para continuar código
  buildContinuationPrompt(userMessage, tokenManager) {
    const model = tokenManager.getModelInfo();
    let prompt = userMessage;

    // Se é IA pequena, incluir contexto anterior
    if (model.isGratuito || model.maxCodeLines < 100) {
      const context = this.buildContextPrompt();
      if (context) {
        prompt = context + "\n\n## NOVA SOLICITAÇÃO\n" + userMessage;
      }
    }

    return prompt;
  }

  // Gerar instruções para fragmentar código
  buildFragmentationInstructions(maxLines) {
    return `

IMPORTANTE: Seu código será usado em um ambiente com limite de tokens.
Se o código for maior que ${maxLines} linhas, divida em partes com comentários claros:

// ===== PARTE 1 =====
// [descrição]
[código]

// ===== PARTE 2 =====
// [descrição]
[código]

Cada parte deve funcionar independentemente ou ser facilmente combinada.`;
  }

  // Resetar cérebro (nova sessão)
  reset() {
    this.memories = [];
    this.codeFragments = [];
  }
}

window.BrainSystem = BrainSystem;

    if (this.models[modelName]) {
      this.currentModel = modelName;
      return true;
    }
    return false;
  }

    return this.models[this.currentModel];
  }

  canMakeRequest(estimatedTokens) {
    const model = this.getModelInfo();
    
    if (this.dailyTokensUsed + estimatedTokens > model.maxTokensPerDay) {
      return {
        allowed: false,
        reason: "Limite diário atingido",
        remaining: model.maxTokensPerDay - this.dailyTokensUsed
      };
    }

    if (estimatedTokens > model.maxTokensPerRequest) {
      return {
        allowed: false,
        reason: "Solicitação muito grande",
        max: model.maxTokensPerRequest,
        requested: estimatedTokens
      };
    }

    return { allowed: true };
  }

  addTokens(inputTokens, outputTokens) {
    const totalTokens = inputTokens + outputTokens;
    this.dailyTokensUsed += totalTokens;
    this.sessionTokensUsed += totalTokens;

    return {
      inputTokens,
      outputTokens,
      totalTokens,
      dailyTotal: this.dailyTokensUsed,
      sessionTotal: this.sessionTokensUsed,
      cost: this.calculateCost(totalTokens)
    };
  }

  calculateCost(tokens) {
    const model = this.getModelInfo();
    return (tokens / 1000) * model.costPer1kTokens;
  }

  estimateTokens(text) {
    // Aproximação: 1 token ≈ 4 caracteres
    return Math.ceil(text.length / 4);
  }

  getProgress() {
    const model = this.getModelInfo();
    const dailyPercent = (this.dailyTokensUsed / model.maxTokensPerDay) * 100;
    
    return {
      model: model.name,
      dailyUsed: this.dailyTokensUsed,
      dailyMax: model.maxTokensPerDay,
      dailyPercent: Math.min(dailyPercent, 100),
      sessionUsed: this.sessionTokensUsed,
      isGratuito: model.isGratuito,
      maxCodeLines: model.maxCodeLines
    };
  }

  reset() {
    this.sessionTokensUsed = 0;
  }
}

        model: "mistralai/Mistral-7B-Instruct-v0.1",
        maxTokens: 2000,
        costPer1k: 0,
        isGratuito: true
      }
    };

    this.currentProvider = "openai-gpt35";
    this.apiKeys = {};
  }

  setProvider(providerKey) {
    if (this.providers[providerKey]) {
      this.currentProvider = providerKey;
      return true;
    }
    return false;
  }

  getProvider() {
    return this.providers[this.currentProvider];
  }

  setApiKey(providerKey, apiKey) {
    this.apiKeys[providerKey] = apiKey;
  }

  getApiKey(providerKey) {
    return this.apiKeys[providerKey] || localStorage.getItem(`ai_key_${providerKey}`);
  }

  async call(messages, systemPrompt = "") {
    const provider = this.getProvider();
    const apiKey = this.getApiKey(this.currentProvider);

    if (!apiKey && provider.type !== "ollama") {
      throw new Error("API Key não configurada para " + provider.name);
    }

    const fullMessages = systemPrompt
      ? [{ role: "system", content: systemPrompt }, ...messages]
      : messages;

    switch (provider.type) {
      case "openai":
        return await this._callOpenAI(fullMessages, provider, apiKey);
      case "claude":
        return await this._callClaude(fullMessages, provider, apiKey);
      case "gemini":
        return await this._callGemini(fullMessages, provider, apiKey);
      case "ollama":
        return await this._callOllama(fullMessages, provider);
      case "groq":
        return await this._callGroq(fullMessages, provider, apiKey);
      case "huggingface":
        return await this._callHuggingFace(fullMessages, provider, apiKey);
      default:
        throw new Error("Provider não suportado");
    }
  }

  async _callOpenAI(messages, provider, apiKey) {
    const response = await fetch("https://api.openai.com/v1/chat/completions", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Authorization": `Bearer ${apiKey}`
      },
      body: JSON.stringify({
        model: provider.model,
        messages,
        max_tokens: provider.maxTokens,
        temperature: 0.7
      })
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error?.message || "Erro OpenAI");
    }

    const data = await response.json();
    return {
      text: data.choices[0].message.content,
      inputTokens: data.usage.prompt_tokens,
      outputTokens: data.usage.completion_tokens,
      totalTokens: data.usage.total_tokens
    };
  }

  async _callClaude(messages, provider, apiKey) {
    const response = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "x-api-key": apiKey,
        "anthropic-version": "2023-06-01"
      },
      body: JSON.stringify({
        model: provider.model,
        max_tokens: provider.maxTokens,
        messages
      })
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error?.message || "Erro Claude");
    }

    const data = await response.json();
    return {
      text: data.content[0].text,
      inputTokens: data.usage.input_tokens,
      outputTokens: data.usage.output_tokens,
      totalTokens: data.usage.input_tokens + data.usage.output_tokens
    };
  }

  async _callGemini(messages, provider, apiKey) {
    const contents = messages.map(msg => ({
      role: msg.role === "assistant" ? "model" : "user",
      parts: [{ text: msg.content }]
    }));

    const response = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/models/${provider.model}:generateContent?key=${apiKey}`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          contents,
          generationConfig: { maxOutputTokens: provider.maxTokens }
        })
      }
    );

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error?.message || "Erro Gemini");
    }

    const data = await response.json();
    const text = data.candidates[0].content.parts[0].text;

    return {
      text,
      inputTokens: Math.ceil(messages.join("").length / 4),
      outputTokens: Math.ceil(text.length / 4),
      totalTokens: Math.ceil((messages.join("") + text).length / 4)
    };
  }

  async _callOllama(messages, provider) {
    const response = await fetch(`${provider.endpoint}/api/chat`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        model: provider.model,
        messages,
        stream: false
      })
    });

    if (!response.ok) {
      throw new Error("Erro Ollama - Certifique-se que está rodando");
    }

    const data = await response.json();
    return {
      text: data.message.content,
      inputTokens: data.prompt_eval_count || 0,
      outputTokens: data.eval_count || 0,
      totalTokens: (data.prompt_eval_count || 0) + (data.eval_count || 0)
    };
  }

  async _callGroq(messages, provider, apiKey) {
    const response = await fetch("https://api.groq.com/openai/v1/chat/completions", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Authorization": `Bearer ${apiKey}`
      },
      body: JSON.stringify({
        model: provider.model,
        messages,
        max_tokens: provider.maxTokens
      })
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error?.message || "Erro Groq");
    }

    const data = await response.json();
    return {
      text: data.choices[0].message.content,
      inputTokens: data.usage.prompt_tokens,
      outputTokens: data.usage.completion_tokens,
      totalTokens: data.usage.total_tokens
    };
  }

  async _callHuggingFace(messages, provider, apiKey) {
    const prompt = messages.map(m => `${m.role}: ${m.content}`).join("\n");

    const response = await fetch(
      `https://api-inference.huggingface.co/models/${provider.model}`,
      {
        method: "POST",
        headers: { "Authorization": `Bearer ${apiKey}` },
        body: JSON.stringify({ inputs: prompt })
      }
    );

    if (!response.ok) {
      throw new Error("Erro HuggingFace");
    }

    const data = await response.json();
    const text = data[0]?.generated_text || "";

    return {
      text,
      inputTokens: Math.ceil(prompt.length / 4),
      outputTokens: Math.ceil(text.length / 4),
      totalTokens: Math.ceil((prompt + text).length / 4)
    };
  }

  listProviders() {
    return Object.entries(this.providers).map(([key, value]) => ({
      key,
      ...value
    }));
  }

  listFreeProviders() {
    return this.listProviders().filter(p => p.isGratuito);
  }

    let supabase = null;

    const $ = (id) => document.getElementById(id);

    async function initSupabase() {
      const url = localStorage.getItem("supabase_url");
      const key = localStorage.getItem("supabase_key");

      if (!url || !key) return false;

      supabase = window.supabase.createClient(url, key);

      const { data: { user } } = await supabase.auth.getUser();
      currentUser = user;

      if (!currentUser) return false;

      await loadConfig();
      return true;
    }

    async function loadConfig() {
      if (!supabase || !currentUser) return;

      const { data } = await supabase
        .from("user_configs")
        .select("*")
        .eq("user_id", currentUser.id)
        .single();

      if (data) {
        $("openaiKey").value = data.openai_key || "";
        $("githubToken").value = data.github_token || "";
        $("googleTtsKey").value = data.google_tts_key || "";
      }
    }

    async function saveConfig() {
      if (!supabase || !currentUser) return;

      const payload = {
        user_id: currentUser.id,
        openai_key: $("openaiKey").value,
        github_token: $("githubToken").value,
        google_tts_key: $("googleTtsKey").value
      };

      const { data: existing } = await supabase
        .from("user_configs")
        .select("id")
        .eq("user_id", currentUser.id)
        .single();

      let result;

      if (existing) {
        result = await supabase
          .from("user_configs")
          .update(payload)
          .eq("user_id", currentUser.id);
      } else {
        result = await supabase
          .from("user_configs")
          .insert(payload);
      }

      if (result.error) {
        showStatus("Erro: " + result.error.message, "error");
        return;
      }

      showStatus("Configuração salva com sucesso!", "success");
    }

    function showStatus(message, type = "success") {

      el.className = "status show " + type;
      el.textContent = message;

      setTimeout(() => {
        el.className = "status";
      }, 3000);

      if (confirm("Limpar todas as configurações?")) {
        $("openaiKey").value = "";
        $("githubToken").value = "";
        $("googleTtsKey").value = "";
